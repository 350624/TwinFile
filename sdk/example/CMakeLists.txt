cmake_minimum_required(VERSION 3.24)
project(TwinFileSelectionExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets)
add_library(TwinFileSelectionExample MODULE example.cpp example.h example.json)
target_include_directories(TwinFileSelectionExample PRIVATE ..)
target_link_libraries(TwinFileSelectionExample PRIVATE Qt6::Widgets)
set_target_properties(TwinFileSelectionExample PROPERTIES PREFIX "")
if(MINGW)
    target_compile_options(TwinFileSelectionExample PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
