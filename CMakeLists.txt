if(EXISTS "${TWINFILE_TOOL_ROOT}/ecj.jar")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/DexBridge.java")
    execute_process(COMMAND "${TWINFILE_TOOL_ROOT}/runtime/bin/java.exe" -jar "${TWINFILE_TOOL_ROOT}/ecj.jar" -17 -encoding UTF-8 -cp "${TWINFILE_TOOL_ROOT}/apktool.jar" -d "${CMAKE_BINARY_DIR}/dex-helper" "${CMAKE_CURRENT_SOURCE_DIR}/assets/DexBridge.java" RESULT_VARIABLE DEX_COMPILE_RESULT)
    if(NOT DEX_COMPILE_RESULT EQUAL 0)
        message(FATAL_ERROR "DEX bridge compilation failed")
    endif()
    qt_add_resources(twinfile_ui dex_resources PREFIX "/dex" BASE "${CMAKE_BINARY_DIR}/dex-helper" FILES "${CMAKE_BINARY_DIR}/dex-helper/DexBridge.class" "${CMAKE_BINARY_DIR}/dex-helper/DexBridge$LimitedWriter.class")
    target_compile_definitions(twinfile_ui PRIVATE TWINFILE_DEX_BRIDGE)
else()
    message(WARNING "ecj.jar not found: Dex editor backend unavailable in this build")
endif()
